  # (1) inter-replicate correlation — find the best pair
  best_cor <- function(wide_df, cols) {
    best_r <- -1; best_p <- 1
    non_nt <- filter(wide_df, !NT)
    for (i in seq_len(length(cols) - 1)) {
      for (j in (i + 1):length(cols)) {
        r <- cor(non_nt[[cols[i]]], non_nt[[cols[j]]])
        p <- cor.test(non_nt[[cols[i]]], non_nt[[cols[j]]])$p.value
        if (r > best_r) { best_r <- r; best_p <- p }
      }

    list(r = best_r, p = best_p)

    type     = curtype,

Here are some functions that I will make use of below.

ranksumROC <- function(x, y, na.rm = TRUE, ...) {
  if (na.rm) {
    x <- x[!is.na(x)]
    y <- y[!is.na(y)]
  }
  curTest        <- wilcox.test(x, y, ...)
  curTest$AUROC  <- (curTest$statistic / length(x)) / length(y)
  curTest
}

Evaluation

The inter-replicate correlations were calculated in the sections above and are stored in metricsBoth. Below, the two remaining metrics are calculated (distinguishing promoter vs other targeting guides, and adjacent vs randomly paired guides).

# (1) similarity between the effect sizes estimated per replicate (above)
metricsBoth <- mutate(metricsBoth, significant = sig < 0.01)

# Other evaluation metrics
allExpts <- distinct(allResultsBoth, celltype, locus, type)

for (ei in seq_len(nrow(allExpts))) {
  curCelltype <- allExpts$celltype[ei]
  curtype     <- allExpts$type[ei]
  curLocus    <- allExpts$locus[ei]

  curResults <- allResultsBoth |>
    filter(celltype == curCelltype, type == curtype, locus == curLocus)

  for (m in unique(curResults$method)) {
    curData <- curResults |>
      filter(method == m, !NT) |>
      arrange(pos)

    # (2) similarity in effect size between adjacent guides
    guideEffectDistances <- tibble(
      method     = m,
      random     = FALSE,
      difference = abs(diff(curData$metric)),
      dist       = abs(diff(curData$pos))
    ) |>
      filter(dist < 100) |>
      select(-dist)

    ### changed 10 to 100 to make this more robust
    curDataSampled <- curData[sample(nrow(curData), size = nrow(curData) * 100, replace = TRUE), ]
    guideEffectDistances <- bind_rows(
      guideEffectDistances,
      tibble(
        method     = m,
        random     = TRUE,
        difference = abs(diff(curDataSampled$metric))
      )
    )

    # random should have more different than adjacent
    curRS <- ranksumROC(
      guideEffectDistances$difference[guideEffectDistances$random],
      guideEffectDistances$difference[!guideEffectDistances$random]
    )
    metricsBoth <- bind_rows(metricsBoth, tibble(
      method      = m, metric = "AUROC-0.5", which = "adjacent_vs_random",
      value       = curRS$AUROC - 0.5, locus = curLocus,
      celltype    = curCelltype, type = curtype,
      sig         = curRS$p.value, significant = curRS$p.value < 0.01
    ))

    # (3) ability to distinguish promoter-targeting guides from other guides.
    if (curtype == "CRISPRi" & !(m %in% c("edgeR", "DESeq2"))) {
      # edgeR and DESeq2 are reversed for CRISPRi
      # non-promoter should have higher effect than promoter (more -ve)
      curRS <- ranksumROC(
        curResults$significance[curResults$method == m & !curResults$NT & !curResults$promoter],
        curResults$significance[curResults$method == m & !curResults$NT &  curResults$promoter]
      )
    } else {
      # promoter should have larger effect than non-promoter
      curRS <- ranksumROC(
        curResults$significance[curResults$method == m & !curResults$NT &  curResults$promoter],
        curResults$significance[curResults$method == m & !curResults$NT & !curResults$promoter]
      )
    }
    metricsBoth <- bind_rows(metricsBoth, tibble(
      method      = m, metric = "AUROC-0.5", which = "promoter_vs_T",
      value       = curRS$AUROC - 0.5, locus = curLocus,
      celltype    = curCelltype, type = curtype,
      sig         = curRS$p.value, significant = curRS$p.value < 0.01
    ))
  }
}

## compile all metrics; label the best in each test and whether any tests were significant (P<0.01)
method_levels <- c("logVsUnsorted", "logHivsLow", "DESeq2", "edgeR", "MAUDE")

metricsBoth2 <- metricsBoth |>
  mutate(method = factor(as.character(method), levels = method_levels)) |>
  group_by(which, locus, type, celltype) |>
  mutate(
    best    = max(value, na.rm = TRUE),
    anySig  = any(significant)
  ) |>
  ungroup() |>
  mutate(
    isBest   = value == best,
    isBestNA = if_else(isBest, TRUE, NA),
    pctOfMax = value / best * 100

make_tile_plot <- function(data, title_str) {
  ggplot(data, aes(x = method, fill = value, y = paste(locus, type, celltype))) +
    geom_tile() +
    geom_text(
